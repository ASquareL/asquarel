📅 Week 8 - Day 3: App Logic Compilation

Duration: 3 hours | Focus: The Engine of the Application

⚙️ The Master Script

This is the final `app.js` file for your Capstone Task Manager. It combines Arrays, Functions, DOM Manipulation, Events, and LocalStorage into one professional architecture.

// 1. SELECT DOM ELEMENTS const input = document.getElementById('taskInput'); const btn = document.getElementById('addTaskBtn'); const list = document.getElementById('taskList'); // 2. INITIALIZE STATE (Load from memory or start empty) let tasks = JSON.parse(localStorage.getItem('tasks')) || []; // 3. RENDER FUNCTION (Draws the UI based on the array) function renderTasks() { list.innerHTML = ''; // Clear old list first tasks.forEach((task, index) => { const li = document.createElement('li'); li.innerHTML = `${task} <button onclick="deleteTask(${index})" style="background: red; border: none; color: white; padding: 5px 10px; cursor: pointer;">X</button>`; list.appendChild(li); }); } // 4. ADD TASK EVENT btn.addEventListener('click', () => { if(input.value === '') return; // Prevent empty tasks tasks.push(input.value); // Add to array localStorage.setItem('tasks', JSON.stringify(tasks)); // Save to memory renderTasks(); // Update the screen input.value = ''; // Clear the input field }); // 5. DELETE TASK FUNCTION // Because this is called directly from the HTML onclick, it must be on the global window window.deleteTask = function(index) { tasks.splice(index, 1); // Remove 1 item at the specific index localStorage.setItem('tasks', JSON.stringify(tasks)); // Update memory renderTasks(); // Update the screen } // 6. BOOT THE APP renderTasks();
